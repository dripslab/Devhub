Quick Start

// 1. Install the SDK
npm install @stellar/stellar-sdk

// 2. Create a Keypair
const StellarSdk = require('@stellar/stellar-sdk');

const pair = StellarSdk.Keypair.random();

console.log("Public Key:", pair.publicKey());
console.log("Secret Key:", pair.secret());
